# Волна 2.16 «Init 2.0 + unpin + память worktree + гигиена доставки» — план реализации

> **Исполнителям:** REQUIRED-СКИЛЛ: используй wolf-sdd (рекомендуется) ИЛИ wolf-execute
> для поэтапной реализации этого плана. Шаги оформлены чекбоксами (`- [ ]`) для трекинга.

**Цель:** первый запуск без барьеров (init без обязательных решений, осторожный повторный init), честная доставка контекста через worktree-границу (common-dir + scope + promote) и гигиена доставки (durability, .gitignore, маскирование секретов, объяснимый пустой call).

**Архитектура:** всё новое — поведение существующих поверхностей `init`/`add`/`call`/`doctor` плюс одна новая CLI-команда `wolf promote`; память `.wolf/` резолвится через git common-dir в main-чекоут, рендеры `.opencode/` остаются локальными; scope — passthrough-поле `worktree:<имя>` / `main`; durability-сплит фиксирует стратегию источников (markdown — истина объектов, events — истина журнала, индекс — производный). Миграций данных нет (P3).

**Стек:** TypeScript, Node 22, vitest, commander; git-метаданные — fs-парс без подпроцессов; полный гейт `npm run check`, e2e — `npm run e2e`.

`docs/superpowers/specs/2026-10-10-wave-2.16-init2-unpin-hygiene-design.md`. Ссылки «§N» — разделы
настоящего плана; «спека §N» — разделы спеки 2.16.

```typescript
// src/adapters/fs/memory-root.ts — целиком
import { existsSync, readFileSync } from 'fs';
import { basename, dirname, isAbsolute, join, resolve } from 'path';

export interface MemoryRootInfo {
  /** Корень чекаута, владеющего общим .wolf/ (main-чекоут; вне worktree — сам baseDir). */
  root: string;
  /** Имя linked worktree или null (main-чекоут / вне git / аномалия). */
  worktree: string | null;
}

/** WP3 §6.1 (спека 2.16): linked worktree резолвит общий .wolf/ в main-чекоуте.
 * Механика спеки — git common-dir (`git rev-parse --git-common-dir`); здесь тот же
 * информационный fs-парс без подпроцессов (прецедент чтения .git/HEAD:
 * heuristic-project-scanner.ts:221-239) — юнит-тестируемо без git, нет подпроцесса
 * в hot-path. Любой сбой — безопасный fallback на локальный .wolf/ (риск Р5 спеки). */
export function resolveMemoryRoot(baseDir: string): MemoryRootInfo {
  try {
    const dotGit = join(baseDir, '.git');
    if (!existsSync(dotGit)) return { root: baseDir, worktree: null };
    const dotGitRaw = readFileSync(dotGit, 'utf-8');
    if (!dotGitRaw.startsWith('gitdir:')) return { root: baseDir, worktree: null }; // обычный чекаут
    const gitdir = resolve(baseDir, dotGitRaw.slice('gitdir:'.length).trim());
    const commondirFile = join(gitdir, 'commondir');
    if (!existsSync(commondirFile)) return { root: baseDir, worktree: null }; // не linked worktree
    const raw = readFileSync(commondirFile, 'utf-8').trim();
    const commonGitDir = isAbsolute(raw) ? raw : resolve(gitdir, raw);
    const root = dirname(commonGitDir);
    if (root === baseDir) return { root: baseDir, worktree: null };
    // bare-репо + linked worktree: commondir указывает мимо чекаута — fallback локальный (Р5)
    if (!existsSync(join(root, '.git'))) return { root: baseDir, worktree: null };
    return { root, worktree: basename(baseDir) };
  } catch {
    return { root: baseDir, worktree: null };
  }
}
```

```typescript
// src/app/use-cases/promote-memory.ts — целиком
import { MemoryStore } from '../../ports/memory-store.port.js';
import { EventLog } from '../../ports/event-log.port.js';
import { Clock } from '../../ports/clock.port.js';
import { IdGenerator } from '../../ports/id-generator.port.js';
import { SearchIndex } from '../../ports/search-index.port.js';
import { MemoryLock } from '../../ports/memory-lock.port.js';
import { UserFacingError } from '../../domain/errors.js';
import type { MemoryObject } from '../../domain/schemas/memory-object-schema.js';

export const MAIN_SCOPE = 'main';

export interface PromoteResult {
  promoted: number;
  scopeFrom: string;
}

/** WP3 §6.2 (спека 2.16): шаг finishing-a-development-branch — scoped-объекты
 * ворктрии становятся main. Автоудаления нет (P3): только перевод scope. */
export async function promoteWorktreeMemory(
  deps: {
    store: MemoryStore;
    log: EventLog;
    clock: Clock;
    idGen: IdGenerator;
    index?: SearchIndex;
    lock?: MemoryLock;
  },
  worktree: string
): Promise<PromoteResult> {
  if (!/^[A-Za-z0-9][A-Za-z0-9._-]*$/.test(worktree)) {
    throw new UserFacingError(`Invalid worktree name: "${worktree}" (see: git worktree list)`);
  }
  const scopeFrom = `worktree:${worktree}`;
  const run = async (): Promise<PromoteResult> => {
    const objects = (await deps.store.list()).filter((o) => (o as { scope?: string }).scope === scopeFrom);
    const now = deps.clock.now();
    let promoted = 0;
    for (const o of objects) {
      await deps.store.update(o.id, { scope: MAIN_SCOPE } as unknown as Partial<MemoryObject>);
      await deps.log.append({
        id: deps.idGen.generateEventId(now),
        type: 'memory.scope_promoted',
        timestamp: now.toISOString(),
        actor: 'system:wolf',
        payload: { memory_id: o.id, from: scopeFrom, to: MAIN_SCOPE },
      });
      if (deps.index) await deps.index.indexObject({ ...o, scope: MAIN_SCOPE } as never);
      promoted += 1;
    }
    return { promoted, scopeFrom };
  };
  return deps.lock ? deps.lock.withLock(run) : run();
}
```

```typescript
// src/app/use-cases/add-memory-object.ts — новые импорты и экспорт
import { isAbsolute, normalize, relative, sep } from 'path';
import { UserFacingError } from '../../domain/errors.js';
import type { MemoryObject } from '../../domain/schemas/memory-object-schema.js';

/** WP3 §6.3: related.files/docs — project-relative posix-пути; абсолют вне root — отказ.
 * Прецедент: tool-librarian.ts:151 (script_path хранится относительным).
 * decisions не трогаем — это id, не пути. */
export function normalizeRelatedPaths(
  related: MemoryObject['related'] | undefined,
  projectRoot: string | undefined
): MemoryObject['related'] {
  const base = related ?? { files: [], docs: [], decisions: [] };
  if (projectRoot === undefined) return base; // legacy-вызовы/фейки без root — как сегодня
  const norm = (p: string): string => {
    const rel = isAbsolute(p) ? relative(projectRoot, p) : normalize(p);
    if (rel.length === 0 || rel.startsWith('..') || isAbsolute(rel)) {
      throw new UserFacingError(`related path escapes the project root: "${p}" (use a project-relative path)`);
    }
    return rel.split(sep).join('/');
  };
  return { files: base.files.map(norm), docs: base.docs.map(norm), decisions: base.decisions };
```

```typescript
// src/app/use-cases/repair-storage.ts — целиком
import { MemoryStore } from '../../ports/memory-store.port.js';
import { EventLog } from '../../ports/event-log.port.js';
import { Clock } from '../../ports/clock.port.js';
import { IdGenerator } from '../../ports/id-generator.port.js';
import { SearchIndex } from '../../ports/search-index.port.js';
import { MemoryLock } from '../../ports/memory-lock.port.js';

export interface RepairReport {
  reindexed: number;
  eventsAppended: number;
  danglingEvents: number;
  /** Человекочитаемый план (dry-run) / лог (apply). */
  actions: string[];
}

export interface RepairDeps {
  store: MemoryStore;
  log: EventLog;
  clock: Clock;
  idGen: IdGenerator;
  index?: SearchIndex;
  /** Сырые непустые строки events.jsonl; не передан — events-часть пропускается. */
  readEventLines?: () => Promise<string[]>;
  lock?: MemoryLock;
}

interface ParsedEvent {
  type?: string;
  payload?: { memory_id?: string };
}

/** WP4 (спека 2.16 §7). Стратегия источников (фиксация планом волны 2.16):
 * markdown — истина объектов; events.jsonl — истина журнала (append-only,
 * недостающие memory.added дозаписываются из markdown); индекс — производный
 * (rebuild). Восстановление идемпотентно; битые строки журнала пропускаются. */
export async function repairStorage(deps: RepairDeps, opts: { dryRun: boolean }): Promise<RepairReport> {
  const run = async (): Promise<RepairReport> => {
    const actions: string[] = [];
    const objects = await deps.store.list();
    const byId = new Map(objects.map((o) => [o.id, o]));

    // 1. Индекс — производный: полный rebuild (идемпотентен).
    let reindexed = 0;
    if (deps.index) {
      actions.push(`index: rebuild from ${objects.length} markdown object(s)`);
      if (!opts.dryRun) await deps.index.rebuild(objects);
      reindexed = objects.length;
    }

    // 2. Журнал — истина: каждой markdown-записи соответствует memory.added.
    let eventsAppended = 0;
    let danglingEvents = 0;
    const lines = deps.readEventLines ? await deps.readEventLines() : [];
    const logged = new Set<string>();
    for (const line of lines) {
      try {
        const e = JSON.parse(line) as ParsedEvent;
        const id = e.payload?.memory_id;
        if (typeof id !== 'string') continue;
        if (e.type === 'memory.added') logged.add(id);
        if (!byId.has(id)) danglingEvents += 1;
      } catch {
        // битая строка журнала — пропускаем; идемпотентный дозапуск завершает восстановление
      }
    }
    if (danglingEvents > 0) {
      actions.push(
        `events: ${danglingEvents} dangling event(s) reference missing objects — warning only (journal is append-only truth)`
      );
    }
    const missing = objects.filter((o) => !logged.has(o.id));
    if (missing.length > 0) {
      actions.push(`events: backfill ${missing.length} missing memory.added event(s) from markdown`);
      if (!opts.dryRun) {
        for (const o of missing) {
          await deps.log.append({
            id: deps.idGen.generateEventId(new Date(o.created_at)),
            type: 'memory.added',
            timestamp: o.created_at,
            actor: o.created_by,
            payload: { memory_id: o.id, type: o.type, ...(o.tags.length > 0 ? { tags: o.tags } : {}) },
          });
        }
      }
      eventsAppended = missing.length;
    }

    if (actions.length === 0) actions.push('storage is consistent — nothing to repair');
    return { reindexed, eventsAppended, danglingEvents, actions };
  };
  return deps.lock ? deps.lock.withLock(run) : run();
}
```

```typescript
// src/app/use-cases/gitignore-block.ts — целиком
import { join } from 'path';

export const GITIGNORE_MARKER = '# wolf-gitignore:v1';

/** WP4 (спека §7): игнорировать машинно-локальное, коммитить память —
 * формула переноса 2.14 §8.3 («память коммитится — клонирование переносит»). */
export function buildGitignoreBlock(): string {
  return [
    GITIGNORE_MARKER,
    '# wolf: machine-local state; .wolf/memory is portable — commit it',
    '.wolf/cache/',
    '.wolf/memory/.lock',
    '.wolf/backup/',
    '# end wolf-gitignore',
    '',
  ].join('\n');
}

export type GitignoreOutcome = 'created' | 'appended' | 'kept';

export async function ensureGitignoreBlock(
  deps: {
    exists: (p: string) => Promise<boolean>;
    readFile: (p: string) => Promise<string | null>;
    writeFile: (p: string, c: string) => Promise<void>;
  },
  baseDir: string
): Promise<GitignoreOutcome> {
  const p = join(baseDir, '.gitignore');
  const current = (await deps.exists(p)) ? await deps.readFile(p) : null;
  if (current === null) {
    await deps.writeFile(p, `${buildGitignoreBlock()}\n`);
    return 'created';
  }
  if (current.includes(GITIGNORE_MARKER)) return 'kept';
  const gap = current.endsWith('\n\n') ? '' : current.endsWith('\n') ? '\n' : '\n\n';
  await deps.writeFile(p, `${current}${gap}${buildGitignoreBlock()}`);
  return 'appended';
}
```
