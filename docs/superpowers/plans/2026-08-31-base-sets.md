> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

```ts
// src/adapters/render/templates-root.ts
import { readFileSync, existsSync } from 'fs';
import { dirname, join } from 'path';
import { fileURLToPath } from 'url';

let cachedRoot: string | null = null;

/** Корень установленного пакета: подъём от текущего модуля до package.json (max 8 уровней). */
export function packageRoot(): string {
  if (cachedRoot) return cachedRoot;
  let dir = dirname(fileURLToPath(import.meta.url));
  for (let i = 0; i < 8; i++) {
    if (existsSync(join(dir, 'package.json'))) {
      cachedRoot = dir;
      return dir;
    }
    dir = dirname(dir);
  }
  throw new Error(`package.json not found upward from ${fileURLToPath(import.meta.url)}`);
}

export function templatesRoot(): string {
  return join(packageRoot(), 'templates', 'base');
}

/** Версия набора для штампов — версия установленного пакета (спека M11). */
export function wolfVersion(): string {
  const pkg = JSON.parse(readFileSync(join(packageRoot(), 'package.json'), 'utf-8')) as { version: string };
  return pkg.version;
}
```

```ts
// src/adapters/render/manifest.ts
export interface HarnessManifest {
  id: string;
  tools: Record<string, string>;
  layout: { agents: string; skills: string; commands: string; plugins: string };
}

export const OPENCODE_MANIFEST: HarnessManifest = {
  id: 'opencode',
  tools: {
    task: 'task',
    skill: 'skill',
    todowrite: 'todowrite',
  },
  layout: {
    agents: '.opencode/agents',
    skills: '.opencode/skills',
    commands: '.opencode/command',
    plugins: '.opencode/plugins',
  },
};

export function substitute(text: string, manifest: HarnessManifest): string {
  return text.replace(/\{\{tool\.(\w+)\}\}/g, (_m, name: string) => {
    const v = manifest.tools[name];
    if (v === undefined) throw new Error(`Unknown tool placeholder: {{tool.${name}}}`);
    return v;
  });
}
```

```ts
// src/adapters/render/stamp.ts
export interface Stamp {
  base: string;
  set: string;
}

function syntax(fileName: string): { open: string; close: string } {
  return /\.(ts|js|mjs|cjs)$/.test(fileName) ? { open: '//', close: '' } : { open: '<!--', close: ' -->' };
}

export function renderStamp(stamp: Stamp, fileName: string): string {
  const s = syntax(fileName);
  return `${s.open} wolf:rendered base=${stamp.base} set=${stamp.set} ${s.close}`.trimEnd();
}

export function parseStamp(text: string): Stamp | null {
  const m = text.match(/(?:<!--|\/\/)\s*wolf:rendered\s+base=(\S+)\s+set=(\S+)(?:\s*-->)?/);
  return m ? { base: m[1], set: m[2] } : null;
}

export function insertStamp(source: string, stamp: Stamp, fileName: string): string {
  const line = renderStamp(stamp, fileName);
  const stripped = source.replace(/(?:<!--|\/\/)\s*wolf:rendered[^\n]*(?:\s*-->)?\n?/g, '');
  if (/^---\n/.test(stripped)) {
    const end = stripped.indexOf('\n---', 4);
    if (end !== -1) {
      const after = stripped.indexOf('\n', end + 1);
      return `${stripped.slice(0, after + 1)}${line}\n${stripped.slice(after + 1)}`;
    }
  }
  return `${line}\n${stripped}`;
}
```

```ts
// src/adapters/render/opencode/opencode-renderer.ts
import { existsSync, readFileSync, readdirSync, rmSync, statSync } from 'fs';
import { join, relative, sep } from 'path';
import { mkdir, writeFile } from 'fs/promises';
import { BaseSetRenderer, RenderOutcome } from '../../../ports/base-set-renderer.port.js';
import { HarnessManifest, substitute } from '../manifest.js';
import { insertStamp, parseStamp } from '../stamp.js';
import { wolfVersion } from '../templates-root.js';

const DIRS = ['agents', 'skills', 'commands', 'plugins'] as const;
/** Соответствие каталог шаблонов → каталог целевого проекта. */
const LAYOUT: Record<(typeof DIRS)[number], string> = {
  agents: '.opencode/agents',
  skills: '.opencode/skills',
  commands: '.opencode/command',
  plugins: '.opencode/plugins',
};

export class OpencodeBaseSetRenderer implements BaseSetRenderer {
  id = 'opencode';

  constructor(
    private readonly templatesBase: string,
    private readonly setVersion: string = wolfVersion(),
    private readonly manifest: HarnessManifest = { id: 'opencode', tools: {}, layout: LAYOUT }
  ) {}

  /** Пару [файл-шаблона → файл-цели]: обход recursive, у skills — <name>/SKILL.md. */
  private *templateFiles(): Generator<{ tplPath: string; targetRel: string; baseName: string }> {
    for (const dir of DIRS) {
      const abs = join(this.templatesBase, dir);
      if (!existsSync(abs)) continue;
      for (const entry of this.walk(abs)) {
        const rel = relative(this.templatesBase, entry);
        const targetRel = rel.startsWith(`skills${sep}`) ? rel.replace(`${sep}SKILL.md`, '/SKILL.md') : rel;
        yield { tplPath: entry, targetRel, baseName: rel.split(sep).join('/') };
      }
    }
  }

  private *walk(dir: string): Generator<string> {
    for (const e of readdirSync(dir)) {
      const p = join(dir, e);
      if (statSync(p).isDirectory()) yield* this.walk(p);
      else yield p;
    }
  }

  private targetPath(targetRel: string): string {
    const [top, ...rest] = targetRel.split(sep);
    return join(this.baseDir0, LAYOUT[top as (typeof DIRS)[number]] ?? top, ...rest);
  }

  private baseDir0 = '';

  async renderBaseSet(baseDir: string): Promise<RenderOutcome[]> {
    this.baseDir0 = baseDir;
    const outcomes: RenderOutcome[] = [];
    for (const { tplPath, targetRel, baseName } of this.templateFiles()) {
      const target = this.targetPath(targetRel);
      if (existsSync(target)) {
        outcomes.push({ file: baseName, action: 'skipped', reason: 'exists (wx-политика)' });
        continue;
      }
      await this.writeRendered(tplPath, target, baseName);
      outcomes.push({ file: baseName, action: 'created' });
    }
    return outcomes;
  }

  async syncBaseSet(baseDir: string): Promise<{ outcomes: RenderOutcome[]; orphaned: string[] }> {
    this.baseDir0 = baseDir;
    const outcomes: RenderOutcome[] = [];
    const seenTargets = new Set<string>();
    for (const { tplPath, targetRel, baseName } of this.templateFiles()) {
      const target = this.targetPath(targetRel);
      seenTargets.add(target);
      if (!existsSync(target)) {
        await this.writeRendered(tplPath, target, baseName);
        outcomes.push({ file: baseName, action: 'created' });
        continue;
      }
      const current = readFileSync(target, 'utf-8');
      if (parseStamp(current) === null) {
        outcomes.push({
          file: baseName,
          action: 'conflict',
          reason: 'unstamped на месте шаблонного — решает владелец',
        });
        continue;
      }
      if (parseStamp(current)?.set === this.setVersion) {
        outcomes.push({ file: baseName, action: 'skipped', reason: 'same set version' });
        continue;
      }
      await this.writeRendered(tplPath, target, baseName);
      outcomes.push({ file: baseName, action: 'updated' });
    }
    const orphaned: string[] = [];
    for (const dir of Object.values(LAYOUT)) {
      const abs = join(baseDir, dir);
      if (!existsSync(abs)) continue;
      for (const f of this.walk(abs)) {
        const rel = relative(baseDir, f);
        if (seenTargets.has(f) && parseStamp(readFileSync(f, 'utf-8')) !== null) continue;
        if (!seenTargets.has(f) && parseStamp(readFileSync(f, 'utf-8')) !== null) orphaned.push(rel);
      }
    }
    return { outcomes, orphaned };
  }

  private async writeRendered(tplPath: string, target: string, baseName: string): Promise<void> {
    const rendered = substitute(readFileSync(tplPath, 'utf-8'), this.manifest);
    const stamped = insertStamp(rendered, { base: baseName, set: this.setVersion }, target.split(sep).pop() ?? '');
    await mkdir(join(target, '..'), { recursive: true });
    await writeFile(target, stamped, 'utf-8');
  }
}
```

```ts
// src/app/use-cases/seed-base-playbooks.ts
import { readFileSync } from 'fs';
import { join } from 'path';

export interface SeedOutcome {
  file: string;
  action: 'created' | 'skipped';
  reason?: string;
}

export interface SeedDeps {
  files: Map<string, string>; // имя файла → содержимое (читает рендерер из templates/base/playbooks/)
  store: { list(): Promise<any[]>; add(o: any): Promise<any> };
  actor: string;
}

function parseFrontmatter(raw: string): { meta: Record<string, string>; body: string } {
  const m = raw.match(/^---\n([\s\S]*?)\n---\n?([\s\S]*)$/);
  if (!m) return { meta: {}, body: raw };
  const meta: Record<string, string> = {};
  for (const line of m[1].split('\n')) {
    const kv = line.match(/^(\w[\w-]*):\s*(.+)$/);
    if (kv) meta[kv[1]] = kv[2].trim();
  }
  return { meta, body: m[2] };
}

/** Посев (спека §7): тег уже есть в памяти → скип; init память не «досеивает». */
export async function seedBasePlaybooks(deps: SeedDeps): Promise<SeedOutcome[]> {
  const existing = await deps.store.list();
  const existingTags = new Set<string>();
  for (const o of existing) for (const t of o.tags ?? []) existingTags.add(t);
  const outcomes: SeedOutcome[] = [];
  for (const [file, raw] of [...deps.files.entries()].sort()) {
    const { meta, body } = parseFrontmatter(raw);
    const tags = (meta.tags ?? '')
      .split(',')
      .map((t) => t.trim())
      .filter(Boolean);
    if (tags.some((t) => existingTags.has(t))) {
      outcomes.push({ file, action: 'skipped', reason: 'тег уже в памяти (скип, R2-M3)' });
      continue;
    }
    await deps.store.add({
      type: 'playbook',
      title: meta.title ?? file,
      body,
      tags,
      confidence: 'high',
      importance: 0.6,
      createdBy: deps.actor,
    });
    for (const t of tags) existingTags.add(t);
    outcomes.push({ file, action: 'created' });
  }
  return outcomes;
}
```
